Configuration Supabase

Exécutez ce script SQL dans votre dashboard Supabase (SQL Editor) pour créer toutes les tables requises :

-- SystemesGED v3.0 — Schema SQL Supabase
-- Exécuter dans : Supabase Dashboard → SQL Editor

create table if not exists users_profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  email text, name text, role text default 'user',
  created_at timestamptz default now()
);

create table if not exists documents (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) on delete cascade,
  name text not null, description text, file_url text,
  file_size bigint default 0, file_type text, storage_path text,
  version_number int default 1, created_at timestamptz default now()
);

create table if not exists document_versions (
  id uuid primary key default gen_random_uuid(),
  document_id uuid references documents(id) on delete cascade,
  version_number int not null, file_url text, file_size bigint,
  storage_path text, created_by uuid, is_active boolean default false,
  created_at timestamptz default now()
);

create table if not exists tags (
  id uuid primary key default gen_random_uuid(),
  name text unique not null, color text default '#3b82f6',
  created_at timestamptz default now()
);

create table if not exists document_tags (
  document_id uuid references documents(id) on delete cascade,
  tag_id uuid references tags(id) on delete cascade,
  primary key (document_id, tag_id)
);

create table if not exists shared_documents (
  id uuid primary key default gen_random_uuid(),
  document_id uuid references documents(id) on delete cascade,
  shared_by uuid references auth.users(id),
  shared_with uuid references auth.users(id),
  shared_with_email text, permission text default 'view',
  expires_at timestamptz, created_at timestamptz default now()
);

create table if not exists workflows (
  id uuid primary key default gen_random_uuid(),
  title text not null, description text, status text default 'pending',
  priority text default 'medium', created_by uuid references auth.users(id),
  document_id uuid references documents(id), due_date date,
  approvers text[], created_at timestamptz default now()
);

create table if not exists activity_logs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) on delete cascade,
  document_id uuid, action text, description text,
  created_at timestamptz default now()
);

-- RLS : activer la sécurité par ligne
alter table users_profiles enable row level security;
alter table documents enable row level security;
alter table document_versions enable row level security;
alter table tags enable row level security;
alter table document_tags enable row level security;
alter table shared_documents enable row level security;
alter table workflows enable row level security;
alter table activity_logs enable row level security;

-- Politiques RLS (chaque user voit ses propres données)
create policy "users see own profile" on users_profiles for all using (auth.uid() = id);
create policy "users manage own documents" on documents for all using (auth.uid() = user_id);
create policy "users manage own versions" on document_versions for all using (auth.uid() = created_by);
create policy "all users see tags" on tags for select using (true);
create policy "authenticated create tags" on tags for insert with check (auth.uid() is not null);
create policy "users delete own tags" on tags for delete using (auth.uid() is not null);
create policy "all see doc_tags" on document_tags for select using (true);
create policy "users manage doc_tags" on document_tags for all using (auth.uid() is not null);
create policy "users see shares" on shared_documents for select using (auth.uid() = shared_by or shared_with_email = (select email from auth.users where id = auth.uid()));
create policy "users create shares" on shared_documents for insert with check (auth.uid() = shared_by);
create policy "users delete own shares" on shared_documents for delete using (auth.uid() = shared_by);
create policy "users manage workflows" on workflows for all using (auth.uid() = created_by);
create policy "users see own logs" on activity_logs for select using (auth.uid() = user_id);
create policy "users create logs" on activity_logs for insert with check (auth.uid() = user_id);

create policy "users delete own logs" on activity_logs for delete using (auth.uid() = user_id);

-- Table RBAC optionnelle (pour persistance des rôles personnalisés)
create table if not exists rbac_roles (
  key text primary key,
  name text not null,
  perm_read boolean default true,
  perm_write boolean default false,
  perm_delete boolean default false,
  perm_users boolean default false,
  perm_logs boolean default false,
  perm_api boolean default false,
  updated_at timestamptz default now()
);
alter table rbac_roles enable row level security;
create policy "all read roles" on rbac_roles for select using (true);
create policy "admin manage roles" on rbac_roles for all using (auth.uid() is not null);

-- Auto-créer le profil à l'inscription
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.users_profiles(id, email, name)
  values (new.id, new.email, new.raw_user_meta_data->>'full_name');
  return new;
end;
$$ language plpgsql security definer;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- Storage bucket
insert into storage.buckets (id, name, public) values ('documents', 'documents', false) on conflict do nothing;
create policy "user upload" on storage.objects for insert with check (bucket_id = 'documents' and auth.uid()::text = (storage.foldername(name))[1]);
create policy "user read own" on storage.objects for select using (bucket_id = 'documents' and auth.uid()::text = (storage.foldername(name))[1]);
create policy "user delete own" on storage.objects for delete using (bucket_id = 'documents' and auth.uid()::text = (storage.foldername(name))[1]);
