create table if not exists users_profiles (id uuid primary key references auth.users(id) on delete cascade, email text, name text, role text default 'user', created_at timestamptz default now());
create table if not exists documents (id uuid primary key default gen_random_uuid(), user_id uuid references auth.users(id) on delete cascade, name text not null, description text, file_url text, file_size bigint default 0, file_type text, storage_path text, version_number int default 1, created_at timestamptz default now());
create table if not exists tags (id uuid primary key default gen_random_uuid(), name text unique not null, color text default '#3b82f6', created_at timestamptz default now());
create table if not exists workflows (id uuid primary key default gen_random_uuid(), title text not null, description text, status text default 'pending', priority text default 'medium', created_by uuid references auth.users(id), document_id uuid references documents(id), due_date date, approvers text[], created_at timestamptz default now());
